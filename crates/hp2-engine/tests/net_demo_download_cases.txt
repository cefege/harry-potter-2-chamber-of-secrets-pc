unsafe fn download_case(index: i32) -> (Box<UNetConnection>, Box<UDownload>) {
    let mut conn = Box::new(unsafe { zeroed::<UNetConnection>() });
    let mut dl = Box::new(unsafe { zeroed::<UDownload>() });
    unsafe { download_receive_file(&SVC, (&mut *dl as *mut UDownload).cast(), (&mut *conn as *mut UNetConnection).cast(), index).unwrap(); }
    (conn, dl)
}

#[test]
fn compressed_signatures_control_codec_stages_and_cleanup_order() {
    for signature in [1234, 5678, 42] {
        reset(); with_mock(|m| { m.signature = signature; m.temp_count = 1; });
        unsafe {
            let (_conn, mut d) = download_case(0); d.transfered = 90; d.is_compressed = 1;
            set_text(&mut d.temp_filename, TEMP);
            download_done(&SVC, (&mut *d as *mut UDownload).cast()).unwrap();
            let e = with_mock(|m| m.events.clone());
            assert_eq!(read_text(&d.temp_filename), "/cache/tmp0002.tmp");
            assert!(position(&e, &format!("delete:{TEMP}")) < position(&e, "archive_delete:-1"));
            assert!(position(&e, "archive_delete:-1") < position(&e, "archive_delete:0"));
            if signature == 42 {
                assert_eq!(read_text(&d.error), "NetSize");
                assert!(!e.iter().any(|s| s == "codec_new" || s == "original_filename"));
                assert!(!contains_trace("move dest="));
            } else {
                assert!(position(&e, "original_filename") < position(&e, "codec_new"));
                assert!(position(&e, "decode") < position(&e, "codec_delete"));
                assert!(position(&e, "codec_delete") < position(&e, &format!("delete:{TEMP}")));
                assert_eq!(e.iter().filter(|s| *s == "rle").count(), if signature == 5678 { 2 } else { 1 });
                assert!(contains_trace("src='/cache/tmp0002.tmp' result=1"));
            }
        }
    }
}

#[test]
fn channel_attachment_payload_and_identity_are_real_leaf_state() {
    reset(); unsafe {
        let mut conn = Box::new(zeroed::<UNetConnection>());
        let mut d = Box::new(zeroed::<UChannelDownload>());
        let p = (&mut *d as *mut UChannelDownload).cast();
        chdl_receive_file(&SVC, p, (&mut *conn as *mut UNetConnection).cast(), 1).unwrap();
        let ch = d.ch.cast::<Channel>(); assert_eq!((*ch).download, p); assert_eq!((*ch).package_index, 1);
        let expected: Vec<u8> = [0x55555555u32,0x66666666,0x77777777,0x88888888].into_iter().flat_map(u32::to_le_bytes).collect();
        assert_eq!(with_mock(|m| m.payloads[0].clone()), expected);
        download_receive_data(&SVC, p, null_mut(), 0).unwrap();
        with_mock(|m| { m.events.clear(); m.bunch_error = true; });
        assert_eq!(chdl_try_skip_file(&SVC, p).unwrap(), 1);
        assert_eq!(with_mock(|m| m.payloads[1].clone()), b"\x05SKIP\0");
        assert!(!with_mock(|m| m.events.iter().any(|s| s == "bunch_is_error")));
        let other = std::ptr::dangling_mut::<c_void>(); (*ch).download = other;
        chdl_destroy(&SVC, p).unwrap(); assert!(d.ch.is_null()); assert_eq!((*ch).download, other);
        drop(Box::from_raw(ch));
    }
}
